<head>

</head>
<body>
    <hr>
    <center>
        <h1><tt>JavaScript_Session</tt></h1>
    </center>
    <hr>
    <fieldset>
        <legend>Checkbox_Validation</legend>
        <center>
            <table border="3">
                <form action=""onsubmit="return Test_Case1()" method="post">
                  <tr>
                    <td><label for="">City | State  | Country</label></td>
                    <td>
                        <select name="" id="A">
                            <option value="">City</option>
                            <option value="">Hyderabad</option>
                            <option value="">Mumbai</option>
                            <option value="">Delhi</option>
                            <option value="">Banguluru</option>
                        </select>
                         <select name="" id="B">
                            <option value="">State</option>
                            <option value="">TG</option>
                            <option value="">MH</option>
                            <option value="">AP</option>
                            <option value="">KA</option>
                        </select>
                         <select name="" id="C">
                            <option value="">Country</option>
                            <option value="">India</option>
                            <option value="">USA</option>
                            <option value="">UK</option>
                            <option value="">Canada</option>
                        </select>
                    </td>
                  </tr>
                  <tr>
                    <td></td>
                    <td>
                         <input type="submit" value="SUBMIT">

            </table>
        </center>
    </fieldset>
    <hr>
    <span id="ABC"></span>
    <hr>
    <script>
        function Test_Case1()
         {
            var select1=document.getElementById("A")
            var select2=document.getElementById("B")
            var select3=document.getElementById("C")
            if(select1.selectedIndex=="")
              {
                 window.alert("Dear User:Please select the city")
                 document.getElementById("ABC").innerHTML="*** Dear User:Please select the city ***"
                 return false

              }
              else if(select2.selectedIndex=="")
                 {
                    window.alert("Dear User:Please select the state")
                    document.getElementById("ABC").innerHTML="*** Dear User:Select the state ***"
                    return false
                 }
                 else if(select3.selectedIndex=="")
                  {
                    window.alert("Dear User:Please select the country")
                    document.getElementById("ABC").innerHTML="*** Dear User:Please select the country ***"
                    return false
                  }
         }
    </script>
    
</body>
</html>
